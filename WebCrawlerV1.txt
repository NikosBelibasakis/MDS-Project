from bs4 import BeautifulSoup
import requests

html_text = requests.get('https://en.wikipedia.org/wiki/List_of_computer_scientists').text
soup = BeautifulSoup(html_text, 'lxml')
list_items = soup.find_all('li',class_='')   #get the <li>s that have no class
for l in list_items:
     a_tags = l.find('a')  #get the text of the first <a> tag in every <li> conent
     print(f'{a_tags} \n\n')
